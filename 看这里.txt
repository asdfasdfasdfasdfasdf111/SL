import Foundation

// MARK: - 数据模型
public struct JavaEnvironment {
    public let path: String      // Java Home 目录
    public let version: String   // 版本号，例如 "1.8.0_202" 或 "17.0.2"
    public let vendor: String?   // 提供者，例如 "Oracle Corporation"
    public let isValid: Bool     // 是否为一个有效的 Java 环境 (可通过 java -version 校验)
}

// MARK: - 主控制器
public final class JavaPathFinder {
    private let fileManager = FileManager.default
    // 缓存标志，确保实例一次运行中，扫描操作只执行一次
    private var cachedEnvironments: [JavaEnvironment]?
    private var cachedPaths: [String]?
    
    public init() {}
    
    /// 获取所有 Java 环境的完整对象数组 (包含版本、提供商等信息)
    public func getAllJavaEnvironments() -> [JavaEnvironment] {
        if let cached = cachedEnvironments { return cached }
        let environments = performFullScan()
        cachedEnvironments = environments
        return environments
    }
    
    /// 获取所有 Java 路径的字符串数组 (性能更高)
    public func getAllJavaPaths() -> [String] {
        if let cached = cachedPaths { return cached }
        let paths = getAllJavaEnvironments().map { $0.path }
        cachedPaths = paths
        return paths
    }
    
    /// 获取推荐的默认 Java (通常会是系统 java_home 指向的版本)
    public func getRecommendedJava() -> JavaEnvironment? {
        let cmd = "/usr/libexec/java_home"
        guard let output = safeExecuteCommand(cmd, args: []),
              !output.isEmpty else { return nil }
        
        let trimmedPath = output.trimmingCharacters(in: .whitespacesAndNewlines)
        // 校验路径是否存在且有 java 可执行文件
        let javaBin = trimmedPath + "/bin/java"
        guard fileManager.fileExists(atPath: javaBin) else { return nil }
        
        return JavaEnvironment(path: trimmedPath,
                               version: getVersionFromPath(trimmedPath) ?? "Unknown",
                               vendor: nil,
                               isValid: true)
    }
    
    // MARK: - 核心扫描逻辑 (组合所有路径源)
    private func performFullScan() -> [JavaEnvironment] {
        var uniquePaths = Set<String>()
        
        // 1. 系统标准路径 (最高优先级)
        let standardPaths = scanDirectory("/Library/Java/JavaVirtualMachines")
        for path in standardPaths { uniquePaths.insert(path) }
        
        // 2. 用户自定义安装路径 (IntelliJ IDEA)
        let jdksPath = NSHomeDirectory() + "/.jdks"
        let customPaths = scanDirectory(jdksPath)
        for path in customPaths { uniquePaths.insert(path) }
        
        // 3. SDKMAN 包管理路径 (含环境变量定位)
        if let sdkmanDir = ProcessInfo.processInfo.environment["SDKMAN_DIR"] {
            let sdkmanPaths = scanDirectory(sdkmanDir + "/candidates/java")
            for path in sdkmanPaths { uniquePaths.insert(path) }
        } else {
            let fallbackSdkman = NSHomeDirectory() + "/.sdkman/candidates/java"
            let sdkmanPaths = scanDirectory(fallbackSdkman)
            for path in sdkmanPaths { uniquePaths.insert(path) }
        }
        
        // 4. Homebrew (Intel / Apple Silicon 兼容)
        let brewPaths = ["/usr/local/opt/openjdk", "/opt/homebrew/opt/openjdk"]
        for brewPath in brewPaths {
            var isDir: ObjCBool = false
            if fileManager.fileExists(atPath: brewPath, isDirectory: &isDir), isDir.boolValue {
                let resolvedPath = resolveSymlink(brewPath) + "/libexec/openjdk.jdk/Contents/Home"
                if fileManager.fileExists(atPath: resolvedPath + "/bin/java") {
                    uniquePaths.insert(resolvedPath)
                }
            }
        }
        
        // 5. HMCL 及其类似启动器路径 (整合请求)
        let hmclPaths = [
            "/Applications/HMCL.app/Contents/runtime",
            NSHomeDirectory() + "/Library/Application Support/hmcl/runtime",
            "/opt/java",          // 通用手动解压位置
            NSHomeDirectory() + "/java"
        ]
        for hmclPath in hmclPaths {
            let scanned = scanDirectory(hmclPath)
            for path in scanned { uniquePaths.insert(path) }
        }
        
        // 6. 额外系统命令获取路径 (java_home -V)
        let javaHomePaths = getPathsFromJavaHomeV()
        for path in javaHomePaths { uniquePaths.insert(path) }
        
        // 组装 JavaEnvironment 对象
        var envs: [JavaEnvironment] = []
        for path in uniquePaths {
            let version = getVersionFromPath(path) ?? "Unknown"
            let isValid = validateJavaHome(path)
            envs.append(JavaEnvironment(path: path, version: version, vendor: nil, isValid: isValid))
        }
        // 去重排序 (按版本降序，使高版本优先)
        envs.sort { compareVersions($0.version, $1.version) == .orderedDescending }
        return envs
    }
    
    // MARK: - 辅助扫描工具 (直接调用命令符，无需图形界面)
    /// 递归扫描指定目录，寻找包含 bin/java 的子目录 (Contents/Home)
    private func scanDirectory(_ basePath: String) -> [String] {
        var results: [String] = []
        guard let enumerator = fileManager.enumerator(atPath: basePath) else { return results }
        for case let path as String in enumerator {
            let fullPath = basePath + "/" + path
            var isDir: ObjCBool = false
            guard fileManager.fileExists(atPath: fullPath, isDirectory: &isDir),
                  isDir.boolValue else { continue }
            
            let javaBinPath = fullPath + "/bin/java"
            if fileManager.fileExists(atPath: javaBinPath) {
                // 如果是标准的 JDK 结构
                let javaHome = fullPath
                results.append(javaHome)
                enumerator.skipDescendants() // 找到后跳过子目录扫描
            } else if path.hasSuffix(".jdk") || path.hasSuffix(".jre") {
                // 检查 Contents/Home/bin/java 结构
                let contentHomeJava = fullPath + "/Contents/Home/bin/java"
                if fileManager.fileExists(atPath: contentHomeJava) {
                    results.append(fullPath + "/Contents/Home")
                    enumerator.skipDescendants()
                }
            }
        }
        return results
    }
    
    /// 执行 /usr/libexec/java_home -V 获取路径
    private func getPathsFromJavaHomeV() -> [String] {
        var paths: [String] = []
        let output = safeExecuteCommand("/usr/libexec/java_home", args: ["-V"])
        guard let output = output else { return paths }
        
        let lines = output.split(separator: "\n")
        for line in lines {
            if line.contains(")") && line.contains(" - ") {
                // 解析类似 "    11.0.2 (x86_64) "Oracle Corporation" - /Library/Java/JavaVirtualMachines/jdk-11.jdk/Contents/Home"
                if let pathStart = line.range(of: "- ") {
                    let pathPart = line[pathStart.upperBound...].trimmingCharacters(in: .whitespaces)
                    paths.append(pathPart)
                }
            }
        }
        return paths
    }
    
    /// 利用 java -version 快速验证 Java Home 是否有效
    private func validateJavaHome(_ path: String) -> Bool {
        let javaBin = path + "/bin/java"
        guard fileManager.fileExists(atPath: javaBin) else { return false }
        let output = safeExecuteCommand(javaBin, args: ["-version"])
        return output != nil // 只要能执行且不报错，就认为是有效的
    }
    
    /// 从路径中提取版本号 (正则匹配)
    private func getVersionFromPath(_ path: String) -> String? {
        let patterns = [
            #"([0-9]+\.?[0-9]*\.?[0-9]*(?:_[0-9]+)?)"#, // 匹配 1.8.0_202 或 17.0.2
            #"jdk[.-]?([0-9\._]+)"#,
            #"openjdk[.-]?([0-9\._]+)"#
        ]
        for pattern in patterns {
            if let regex = try? NSRegularExpression(pattern: pattern),
               let match = regex.firstMatch(in: path, range: NSRange(path.startIndex..., in: path)) {
                let versionRange = Range(match.range(at: 1), in: path)!
                return String(path[versionRange])
            }
        }
        return nil
    }
    
    /// 解析符号链接
    private func resolveSymlink(_ path: String) -> String {
        let resolvedPath = fileManager.destinationOfSymbolicLink(atPath: path) ?? path
        return resolvedPath
    }
    
    /// 安全执行命令并捕获输出 (性能优化点: 不创建额外线程)
    private func safeExecuteCommand(_ execPath: String, args: [String]) -> String? {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: execPath)
        process.arguments = args
        
        let outputPipe = Pipe()
        let errorPipe = Pipe()
        process.standardOutput = outputPipe
        process.standardError = errorPipe
        
        do {
            try process.run()
            process.waitUntilExit()
            guard process.terminationStatus == 0 else { return nil }
            
            let outputData = outputPipe.fileHandleForReading.readDataToEndOfFile()
            return String(data: outputData, encoding: .utf8)?.trimmingCharacters(in: .whitespacesAndNewlines)
        } catch {
            return nil
        }
    }
}

// MARK: - 版本比较工具
private func compareVersions(_ v1: String, _ v2: String) -> ComparisonResult {
    let components1 = v1.split(separator: ".", omittingEmptySubsequences: false)
    let components2 = v2.split(separator: ".", omittingEmptySubsequences: false)
    
    for (c1, c2) in zip(components1, components2) {
        if let n1 = Int(c1), let n2 = Int(c2) {
            if n1 != n2 {
                return n1 < n2 ? .orderedAscending : .orderedDescending
            }
        } else {
            if c1 != c2 {
                return c1 < c2 ? .orderedAscending : .orderedDescending
            }
        }
    }
    return components1.count == components2.count ? .orderedSame : (components1.count < components2.count ? .orderedAscending : .orderedDescending)
}

// JavaInstaller.swift
// 功能：自动下载并安装 Microsoft Build of OpenJDK
// 使用方式：调用 JavaInstaller.install(version: "25.0.3") 或最新版本
// 环境要求：macOS 10.15+，需要网络连接
import Foundation

enum JavaInstallerError: Error, LocalizedError {
    case unsupportedArchitecture
    case invalidVersion
    case downloadFailed(underlying: Error? = nil)
    case installationFailed(exitCode: Int32, message: String? = nil)
    case tempDirectoryCreationFailed
    
    var errorDescription: String? {
        switch self {
        case .unsupportedArchitecture:
            return "不支持的 CPU 架构（仅支持 Intel x86_64 和 Apple Silicon arm64）"
        case .invalidVersion:
            return "版本号格式无效，请使用如 \"25.0.3\" 的格式"
        case .downloadFailed(let underlying):
            return "下载失败\(underlying.map { ": \($0.localizedDescription)" } ?? "")"
        case .installationFailed(let exitCode, let message):
            return "安装失败 (退出码 \(exitCode))\(message.map { ": \($0)" } ?? "")"
        case .tempDirectoryCreationFailed:
            return "无法创建临时目录"
        }
    }
}

/// Java 安装器
class JavaInstaller {
    
    /// 安装指定版本的 Microsoft OpenJDK
    /// - Parameter version: 版本号，例如 "25.0.3"、"21.0.2"，默认为最新稳定版
    /// - Throws: JavaInstallerError
    static func install(version: String = "25.0.3") throws {
        let arch = try detectArchitecture()
        let downloadURL = try makeDownloadURL(version: version, arch: arch)
        print("⚙️ 开始下载 Java \(version) for \(arch) ...")
        
        // 1. 下载 .pkg 到临时目录
        let tempDir = FileManager.default.temporaryDirectory.appendingPathComponent(UUID().uuidString)
        try FileManager.default.createDirectory(at: tempDir, withIntermediateDirectories: true)
        let pkgPath = tempDir.appendingPathComponent("openjdk-\(version)-\(arch).pkg")
        
        do {
            try downloadFile(from: downloadURL, to: pkgPath)
            print("✅ 下载完成: \(pkgPath.path)")
            
            // 2. 安装 .pkg（请求管理员权限）
            try installPackage(at: pkgPath)
            print("✅ Java \(version) 安装成功")
        } catch {
            throw error
        } finally {
            // 清理临时文件
            try? FileManager.default.removeItem(at: tempDir)
        }
    }
    
    // MARK: - Private Helpers
    
    /// 检测当前 Mac 的架构，返回用于下载链接的标识符
    private static func detectArchitecture() throws -> String {
        var sysinfo = utsname()
        uname(&sysinfo)
        let machine = String(cString: &sysinfo.machine.0)
        switch machine {
        case "x86_64":
            return "x64"
        case "arm64":
            return "aarch64"
        default:
            throw JavaInstallerError.unsupportedArchitecture
        }
    }
    
    /// 根据版本和架构构造下载 URL（遵循微软官方链接规律）
    private static func makeDownloadURL(version: String, arch: String) throws -> URL {
        // 版本号必须符合 x.y.z 格式
        let versionPattern = #"^\d+\.\d+\.\d+$"#
        guard version.range(of: versionPattern, options: .regularExpression) != nil else {
            throw JavaInstallerError.invalidVersion
        }
        let urlString = "https://aka.ms/download-jdk/microsoft-jdk-\(version)-macos-\(arch).pkg"
        guard let url = URL(string: urlString) else {
            throw JavaInstallerError.invalidVersion
        }
        return url
    }
    
    /// 使用 URLSession 下载文件（同步等待）
    private static func downloadFile(from url: URL, to destination: URL) throws {
        let semaphore = DispatchSemaphore(value: 0)
        var downloadError: Error?
        
        let task = URLSession.shared.downloadTask(with: url) { tempURL, response, error in
            defer { semaphore.signal() }
            if let error = error {
                downloadError = error
                return
            }
            guard let tempURL = tempURL else {
                downloadError = URLError(.badServerResponse)
                return
            }
            do {
                if FileManager.default.fileExists(atPath: destination.path) {
                    try FileManager.default.removeItem(at: destination)
                }
                try FileManager.default.moveItem(at: tempURL, to: destination)
            } catch {
                downloadError = error
            }
        }
        task.resume()
        _ = semaphore.wait(timeout: .now() + 300) // 5 分钟超时
        
        if let error = downloadError {
            throw JavaInstallerError.downloadFailed(underlying: error)
        }
    }
    
    /// 以管理员权限安装 .pkg 文件
    private static func installPackage(at pkgPath: URL) throws {
        // 使用 AppleScript 请求管理员权限并运行 installer 命令
        let script = """
        do shell script "installer -pkg '\(pkgPath.path)' -target /" with administrator privileges
        """
        var error: NSDictionary?
        guard let scriptObject = NSAppleScript(source: script) else {
            throw JavaInstallerError.installationFailed(exitCode: -1, message: "无法创建 AppleScript")
        }
        let output = scriptObject.executeAndReturnError(&error)
        if let errorDict = error {
            let errorMsg = errorDict.description
            throw JavaInstallerError.installationFailed(exitCode: -1, message: errorMsg)
        }
        // 检查 installer 输出（可选）
        let resultString = output.stringValue ?? ""
        if resultString.contains("The install was successful") == false &&
           resultString.contains("成功") == false {
            // 并非严格必须包含成功关键字，但可给出警告
            print("⚠️ 安装输出未包含明确成功标志，请手动验证: \(resultString)")
        }
        // 退出码 0 表示成功，AppleScript 执行失败时抛异常，所以此处默认成功
    }
}
